using System;
using UnityEngine;

    #ROOTNAMESPACEBEGIN#
//Send the comparison operator to use as a BitFlag in the Comparable attribute 
//and implement the necessary functionality directly in the Execute function.
[Serializable, Readable, Comparable(Comparison.EQ | Comparison.NE)]
public class #SCRIPTNAME# : ConditionModule<>
{
    public override bool Execute(Comparison comparison)
    {
        switch (comparison)
        {
            case Comparison.EQ: return false; 
                
            case Comparison.NE: return true;
        }
    }
}
